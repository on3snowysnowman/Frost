442 void _remove_entity_from_tilemap(const entity_id id, uint16_t x_position, uint16_t y_position)
443 {
444 EntityTracker* targ_tracker {};
445
446 // Iterator to the beginning of the target list at the entity's position.
447 typename std::list<T>::const_iterator it_begin {};
448
449 // Iterator to the end of the target list at the entity's position.
450 typename std::list<T>::const_iterator it_end {};
451
452 try
453 {
454 // Get the tracker assigned to this ID.
455 targ_tracker = &all_entities.at(id);
456
457 it_begin = contents.at(y_position).at(x_position).begin();
458 it_end = contents.at(y_position).at(x_position).end();
459 }
460
461 // If an error ocurred when removing the entity, handle it.
462 catch(const std::out_of_range& e)
463 {
464 std::string error_message = e.what();
465
466 // If the entity's ID is not a valid one.
467 if(!_does_id_exist(id))
468 {
469 error_message = "Invalid entity ID: " + std::to_string(id);
470 }
471
472 // The x position is out of bounds.
473 if(x_position >= N)
474 {
475 error_message = "Invalid x position: " + std::to_string(x_position);
476 }
477
478 // The y position is out of bounds.
479 else if(y_position >= N)
480 {
481 error_message = "Invalid y position: " + std::to_string(y_position);
482 }
483
484 // Output the exception to the crash log file and softly crash.
485 OUTPUT_CRASH_DETAILS(error_message);
486 exit(1);
487 }
488
489 // Iterate over each entity_id in the list at the position.
490 while(it_begin != it_end)
491 {
492 // If the parsed entity is the entity to be removed.
493 if(*it_begin == targ_tracker->entity)
494 {
495 // Erase the entity at the iterator's position.
496 contents.at(y_position).at(x_position).erase(it_begin);
497 return;
498 }
499
500 ++it_begin;
501 }
502
503 // If this part of the function is reached, the entity was not found. Output an error and
504 // softly crash the program.
505
506 OUTPUT_CRASH_DETAILS("Entity was requested to be removed, but could not be found.");
507 }
